Table attendee {
  attendee_id integer [primary key, increment]
  name varchar(255) [not null]
  email varchar(255) [unique, not null]
  phone varchar(50)
  password varchar(255) [not null]
  registration_date timestamp [default: `CURRENT_TIMESTAMP`]
  status varchar(50) [default: 'active']
  profile_photo varchar(500)
  payment_fee decimal(10,2) [default: 200.00]
  password_changed boolean [default: true]
  is_temporary_password boolean [default: false]
  
  Note: 'Stores exhibition attendee/visitor information'
}

Table exhibitor {
  exhibitor_id integer [primary key, increment]
  company_name varchar(255) [not null]
  contact_person varchar(255)
  email varchar(255) [unique, not null]
  booth_number varchar(50)
  product_ids text
  logo_url varchar(500)
  floor_number varchar(10)
  password varchar(255)
  password_changed boolean [default: false]
  is_temporary_password boolean [default: true]
  status varchar(50) [default: 'active']
  
  Note: 'Stores exhibitor company information and booth assignments'
}

Table product {
  product_id integer [primary key, increment]
  name varchar(255) [not null]
  description text
  category varchar(100)
  exhibitor_id integer [ref: > exhibitor.exhibitor_id, note: 'ON DELETE CASCADE']
  image_url varchar(500)
  status varchar(50) [default: 'active']
  
  Note: 'Products/services displayed by exhibitors'
}

Table conference {
  conference_id integer [primary key, increment]
  title varchar(255) [not null]
  description text
  date date [not null]
  time varchar(50)
  location varchar(255)
  speaker varchar(255)
  floor_number varchar(10)
  
  Note: 'Conference sessions and events'
}

Table speaker {
  speaker_id integer [primary key, increment]
  name varchar(255) [not null]
  bio text
  photo_url varchar(500)
  expertise varchar(255)
  conference_id integer [ref: > conference.conference_id, note: 'ON DELETE SET NULL']
  
  Note: 'Conference speakers and their information'
}

Table sponsor {
  sponsor_id integer [primary key, increment]
  company_name varchar(255) [not null]
  logo_url varchar(500)
  website varchar(500)
  sponsorship_level varchar(100)
  status varchar(50) [default: 'active']
  
  Note: 'Exhibition sponsors (Gold, Silver, Bronze, etc.)'
}

Table partner {
  partner_id integer [primary key, increment]
  company_name varchar(255) [not null]
  logo_url varchar(500)
  website varchar(500)
  partnership_type varchar(100)
  status varchar(50) [default: 'active']
  
  Note: 'Exhibition partners and collaborators'
}

Table floor {
  floor_id integer [primary key, increment]
  floor_number integer [unique, not null]
  layout_image varchar(500)
  exhibitor_ids text [note: 'Comma-separated exhibitor IDs']
  conference_ids text [note: 'Comma-separated conference IDs']
  
  Note: 'Exhibition floor layouts and mappings'
}

// Relationships
// Ref: exhibitor.exhibitor_id < product.exhibitor_id [delete: cascade]
// Ref: conference.conference_id < speaker.conference_id [delete: set null]

// Note: Floor relationships are managed via text fields (exhibitor_ids, conference_ids)
// Note: Attendee, Sponsor, and Partner are independent entities with no foreign keys

# CROW'S FOOT NOTATION (Text-based)
# ============================================================================

EXHIBITOR ||--o{ PRODUCT
  One exhibitor has zero or many products
  One product belongs to exactly one exhibitor
  Relationship: exhibitor_id (FK in PRODUCT)
  Cascade: ON DELETE CASCADE

CONFERENCE ||--o{ SPEAKER
  One conference has zero or many speakers
  One speaker belongs to zero or one conference
  Relationship: conference_id (FK in SPEAKER)
  Cascade: ON DELETE SET NULL

FLOOR ||--o{ EXHIBITOR
  One floor contains zero or many exhibitors
  One exhibitor is on exactly one floor
  Relationship: floor_number (text field in EXHIBITOR)
  Tracking: exhibitor_ids (text field in FLOOR)

FLOOR ||--o{ CONFERENCE
  One floor hosts zero or many conferences
  One conference is on exactly one floor
  Relationship: floor_number (text field in CONFERENCE)
  Tracking: conference_ids (text field in FLOOR)

ATTENDEE (Independent)
  No foreign key relationships
  Standalone entity for visitor registration

SPONSOR (Independent)
  No foreign key relationships
  Standalone entity for sponsor management

PARTNER (Independent)
  No foreign key relationships
  Standalone entity for partner management

# ============================================================================
# RELATIONSHIP SUMMARY
# ============================================================================

Direct Relationships (with Foreign Keys):
1. EXHIBITOR → PRODUCT (1:N) - One exhibitor has many products
2. CONFERENCE → SPEAKER (1:N) - One conference has many speakers

Indirect Relationships (via text fields):
3. FLOOR → EXHIBITOR (1:N) - One floor contains many exhibitors
4. FLOOR → CONFERENCE (1:N) - One floor hosts many conferences

Independent Entities (no relationships):
5. ATTENDEE - Visitor registration
6. SPONSOR - Exhibition sponsors
7. PARTNER - Exhibition partners
